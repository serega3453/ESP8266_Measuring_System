# Скачать и установить Python:
https://www.python.org/downloads/windows/
# При установке включить галочку:
Add Python to PATH

# Создать папку, например:
C:\udp_logger

# Поместить в неё udp_logger.py

# Открыть PowerShell в этой папке:
Shift + ПКМ → "Открыть окно PowerShell здесь"

# Запустить логгер
python udp_logger.py

# Логи будут записываться в thermal.log
